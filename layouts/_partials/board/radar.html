{{- /* Terminal radar scope under the board, from the Stitch home-page study (not the design artboards). Contacts are the strips in the bays named by params.radar.bays (ACTIVE by default). Hugo has no layout engine to place blips, so each position is computed here: contacts are spread evenly round the scope in callsign order, and a hash of the callsign adds a little angle jitter and picks the range. The same content gives the same picture on every build, with no JavaScript. The scope repeats the bay, so it is hidden from assistive technology and its links are out of the tab order; the strips carry the same links. */ -}} {{- $cfg := site.Params.radar | default dict }} {{- if ne (index $cfg "enable") false }} {{- $want := slice }} {{- range (index $cfg "bays" | default (slice "ACTIVE")) }}{{ $want = $want | append (upper .) }}{{ end }} {{- $contacts := slice }} {{- $labels := slice }} {{- range .bays }} {{- if in $want .name }} {{- $contacts = $contacts | append .strips }} {{- $labels = $labels | append (T (printf "status%s" .name) | default .name) }} {{- end }} {{- end }} {{- $contacts = sort $contacts "Params.callsign" }} {{- $n := len $contacts }} {{- $bayLabel := delimit $labels " / " }}
